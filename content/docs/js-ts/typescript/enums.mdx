---
title: "Enums"
description: "Numeric, string and const enums, and the as const alternative."
---

```typescript
// Numeric enum (default)
enum Direction {
  Up,    // 0
  Down,  // 1
  Left,  // 2
  Right  // 3
}

Direction.Up;   // 0
Direction[0];   // 'Up' (reverse mapping)

// String enum (preferred — more debuggable)
enum Status {
  Pending = 'PENDING',
  Active = 'ACTIVE',
  Inactive = 'INACTIVE',
}

Status.Pending; // 'PENDING'

// Const enum — inlined at compile time (no JS object generated)
const enum Size {
  Small = 1,
  Medium = 2,
  Large = 3,
}
// Compiled: uses values directly (e.g., 1 instead of Size.Small)

// Alternative: use const object (recommended over enum by many)
const STATUS = {
  Pending: 'PENDING',
  Active: 'ACTIVE',
} as const;

type Status = typeof STATUS[keyof typeof STATUS];
// 'PENDING' | 'ACTIVE'
```
