---
title: "2.16 Custom error classes"
---

Domain errors make services readable and let controllers map errors to HTTP codes without string matching.

```ts
abstract class DomainError extends Error {
  abstract readonly code: string;
  constructor(message: string, options?: ErrorOptions) {
    super(message, options);
    this.name = new.target.name; // "InsufficientFundsError", etc.
  }
}

class InsufficientFundsError extends DomainError {
  readonly code = "INSUFFICIENT_FUNDS";
  constructor(
    readonly accountId: string,
    readonly requiredPaise: number,
    readonly availablePaise: number,
  ) {
    super(`account ${accountId} needs ${requiredPaise} paise, has ${availablePaise}`);
  }
}

class OrderNotFoundError extends DomainError {
  readonly code = "ORDER_NOT_FOUND";
  constructor(readonly orderId: string) {
    super(`order ${orderId} not found`);
  }
}

class PaymentGatewayError extends DomainError {
  readonly code = "GATEWAY_ERROR";
}

class Wallet {
  constructor(readonly id: string, private balancePaise: number) {}
  debit(paise: number): void {
    if (paise > this.balancePaise) throw new InsufficientFundsError(this.id, paise, this.balancePaise);
    this.balancePaise -= paise;
  }
}

function charge(): void {
  try {
    throw new Error("socket hang up");
  } catch (cause) {
    throw new PaymentGatewayError("razorpay call failed", { cause });
  }
}

// Map errors to HTTP status in one place
function toHttp(err: unknown): string {
  if (err instanceof InsufficientFundsError) return `402 ${err.code} short by ${err.requiredPaise - err.availablePaise}`;
  if (err instanceof OrderNotFoundError) return `404 ${err.code}`;
  if (err instanceof DomainError) {
    const cause = err.cause instanceof Error ? ` (cause: ${err.cause.message})` : "";
    return `502 ${err.name}: ${err.message}${cause}`;
  }
  return "500 INTERNAL";
}

const wallet = new Wallet("W1", 10_000);
const attempts: Array<() => void> = [
  () => wallet.debit(25_000),
  () => { throw new OrderNotFoundError("O-404"); },
  charge,
  () => { throw "a string, not an Error"; },
];

for (const run of attempts) {
  try {
    run();
  } catch (err) {
    console.log(toHttp(err));
  }
}

const e = new InsufficientFundsError("W1", 1, 0);
console.log(`${e.name} | instanceof Error=${e instanceof Error} | ${e.message}`);
```

```text
402 INSUFFICIENT_FUNDS short by 15000
404 ORDER_NOT_FOUND
502 PaymentGatewayError: razorpay call failed (cause: socket hang up)
500 INTERNAL
InsufficientFundsError | instanceof Error=true | account W1 needs 1 paise, has 0
```

Notes:

- Set `this.name`, otherwise logs show plain `Error`. `new.target.name` does it once in the base class.
- With a modern target (ES2015+), `instanceof` works for subclasses of `Error`. The old `Object.setPrototypeOf(this, X.prototype)` fix is only needed when compiling down to ES5.
- Use `{ cause }` to wrap low-level errors without losing them.
- In a `catch`, the error is `unknown` (with `strict`). Narrow it before use.
