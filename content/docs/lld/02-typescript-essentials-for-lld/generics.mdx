## Intuition

Generics let you write a component once and keep full type information for each use. In LLD the most common generic pieces are repositories, caches, event buses, and result types.

## A generic in-memory repository

```ts
interface Entity {
  readonly id: string;
}

interface Repository<T extends Entity> {
  save(entity: T): T;
  findById(id: string): T | undefined;
  findAll(predicate?: (e: T) => boolean): T[];
  delete(id: string): boolean;
}

class InMemoryRepository<T extends Entity> implements Repository<T> {
  private readonly store = new Map<string, T>();

  save(entity: T): T {
    this.store.set(entity.id, entity);
    return entity;
  }
  findById(id: string): T | undefined {
    return this.store.get(id);
  }
  findAll(predicate: (e: T) => boolean = () => true): T[] {
    return [...this.store.values()].filter(predicate);
  }
  delete(id: string): boolean {
    return this.store.delete(id);
  }
}

interface Stock extends Entity { symbol: string; ltpPaise: number }
interface Customer extends Entity { name: string; segment: "RETAIL" | "HNI" }

const stocks = new InMemoryRepository<Stock>();
stocks.save({ id: "S1", symbol: "INFY", ltpPaise: 152_050 });
stocks.save({ id: "S2", symbol: "TCS", ltpPaise: 389_000 });

const customers = new InMemoryRepository<Customer>();
customers.save({ id: "C1", name: "Ritesh", segment: "RETAIL" });

const pricey = stocks.findAll((s) => s.ltpPaise > 200_000).map((s) => s.symbol);
console.log(`pricey: ${pricey.join(",")}`);
console.log(`customer: ${customers.findById("C1")?.name}`);
console.log(`deleted S1: ${stocks.delete("S1")}, left: ${stocks.findAll().length}`);
// customers.save({ id: "C2", symbol: "X", ltpPaise: 1 }); // ❌ error: not a Customer
// new InMemoryRepository<{ name: string }>();           // ❌ error: missing 'id' (constraint)
```

```text
pricey: TCS
customer: Ritesh
deleted S1: true, left: 1
```

## Constraints, `keyof`, and defaults

```ts
// Constraint with keyof: K must be a real key of T
function pluck<T, K extends keyof T>(items: readonly T[], key: K): T[K][] {
  return items.map((i) => i[key]);
}

// Generic default: most callers want string errors
type Result<T, E = string> =
  | { ok: true; value: T }
  | { ok: false; error: E };

function parseQty(raw: string): Result<number> {
  const n = Number(raw);
  return Number.isInteger(n) && n > 0 ? { ok: true, value: n } : { ok: false, error: `bad qty "${raw}"` };
}

// Generic class with a default type parameter
class EventBus<TEvents extends Record<string, unknown> = Record<string, unknown>> {
  private handlers = new Map<keyof TEvents, Array<(p: never) => void>>();

  on<K extends keyof TEvents>(event: K, handler: (payload: TEvents[K]) => void): void {
    const list = this.handlers.get(event) ?? [];
    list.push(handler as (p: never) => void);
    this.handlers.set(event, list);
  }

  emit<K extends keyof TEvents>(event: K, payload: TEvents[K]): void {
    for (const h of this.handlers.get(event) ?? []) {
      (h as (p: TEvents[K]) => void)(payload);
    }
  }
}

type TradingEvents = {
  orderPlaced: { orderId: string; qty: number };
  priceTick: { symbol: string; ltpPaise: number };
};

const bus = new EventBus<TradingEvents>();
bus.on("orderPlaced", (p) => console.log(`placed ${p.orderId} x${p.qty}`));
bus.on("priceTick", (p) => console.log(`tick ${p.symbol} ${p.ltpPaise}`));
bus.emit("orderPlaced", { orderId: "O-7", qty: 25 });
bus.emit("priceTick", { symbol: "INFY", ltpPaise: 152_100 });
// bus.emit("priceTick", { orderId: "x" }); // ❌ error: wrong payload for event

const rows = [{ symbol: "INFY", qty: 10 }, { symbol: "TCS", qty: 3 }];
console.log(pluck(rows, "symbol").join(","));
// pluck(rows, "price"); // ❌ error: "price" is not a key

for (const raw of ["15", "-2"]) {
  const r = parseQty(raw);
  console.log(r.ok ? `qty=${r.value}` : `error: ${r.error}`);
}
```

```text
placed O-7 x25
tick INFY 152100
INFY,TCS
qty=15
error: bad qty "-2"
```

Rules of thumb:

- Add a type parameter only if it **links** two things (input to output, key to value). A generic used once is just noise.
- Constrain as little as needed (`T extends Entity`), so the component stays reusable.
- Inside a generic body you know only what the constraint says. If you find yourself casting a lot, the constraint is probably wrong.

> **How to say it in an interview:** "I'll make the repository generic over `T extends Entity`, so every aggregate gets the same CRUD with full typing, and the storage can be swapped later for a DB-backed implementation of the same interface."
