This is the single most useful TypeScript feature for LLD. It lets you model a **state machine** where each state carries only the data that makes sense for it.

## Modelling order states

```ts
type OrderState =
  | { kind: "PENDING"; createdAt: number }
  | { kind: "OPEN"; exchangeOrderId: string }
  | { kind: "PARTIALLY_FILLED"; exchangeOrderId: string; filledQty: number }
  | { kind: "FILLED"; avgPricePaise: number }
  | { kind: "CANCELLED"; reason: string }
  | { kind: "REJECTED"; reason: string };

function assertNever(x: never): never {
  throw new Error(`unhandled state: ${JSON.stringify(x)}`);
}

function describe(s: OrderState, totalQty: number): string {
  switch (s.kind) {
    case "PENDING":
      return `pending since t=${s.createdAt}`;
    case "OPEN":
      return `open on exchange as ${s.exchangeOrderId}`;
    case "PARTIALLY_FILLED":
      return `${s.filledQty}/${totalQty} filled (${s.exchangeOrderId})`;
    case "FILLED":
      return `filled @ ₹${(s.avgPricePaise / 100).toFixed(2)}`;
    case "CANCELLED":
    case "REJECTED":
      return `${s.kind.toLowerCase()}: ${s.reason}`; // both have `reason`
    default:
      return assertNever(s); // compile error here if a new state is added and not handled
  }
}

// Transitions as a pure function: the event type is also a discriminated union
type OrderEvent =
  | { type: "ACK"; exchangeOrderId: string }
  | { type: "FILL"; qty: number; pricePaise: number }
  | { type: "CANCEL"; reason: string };

function transition(s: OrderState, e: OrderEvent, totalQty: number): OrderState {
  switch (e.type) {
    case "ACK":
      if (s.kind !== "PENDING") throw new Error(`cannot ACK in ${s.kind}`);
      return { kind: "OPEN", exchangeOrderId: e.exchangeOrderId };
    case "FILL": {
      if (s.kind !== "OPEN" && s.kind !== "PARTIALLY_FILLED") throw new Error(`cannot FILL in ${s.kind}`);
      const already = s.kind === "PARTIALLY_FILLED" ? s.filledQty : 0;
      const filled = already + e.qty;
      if (filled > totalQty) throw new Error("overfill");
      return filled === totalQty
        ? { kind: "FILLED", avgPricePaise: e.pricePaise }
        : { kind: "PARTIALLY_FILLED", exchangeOrderId: s.exchangeOrderId, filledQty: filled };
    }
    case "CANCEL":
      if (s.kind === "FILLED" || s.kind === "CANCELLED" || s.kind === "REJECTED") {
        throw new Error(`cannot CANCEL in ${s.kind}`);
      }
      return { kind: "CANCELLED", reason: e.reason };
    default:
      return assertNever(e);
  }
}

const events: OrderEvent[] = [
  { type: "ACK", exchangeOrderId: "NSE-881" },
  { type: "FILL", qty: 4, pricePaise: 152_000 },
  { type: "FILL", qty: 6, pricePaise: 152_050 },
  { type: "CANCEL", reason: "too late" },
];

let state: OrderState = { kind: "PENDING", createdAt: 0 };
console.log(describe(state, 10));
for (const e of events) {
  try {
    state = transition(state, e, 10);
    console.log(describe(state, 10));
  } catch (err) {
    if (err instanceof Error) console.log(`error: ${err.message}`);
  }
}
```

```text
pending since t=0
open on exchange as NSE-881
4/10 filled (NSE-881)
filled @ ₹1520.50
error: cannot CANCEL in FILLED
```

## Why `never` catches missing cases

After every `case` has returned, TypeScript narrows `s` to `never` in the `default` branch. If someone adds `{ kind: "EXPIRED" }` to the union but forgets the `switch`, `s` in `default` has type `{ kind: "EXPIRED" }`, which is not assignable to `never`:

```ts
// ❌ after adding | { kind: "EXPIRED" } to OrderState:
default:
  return assertNever(s);
  // error: Argument of type '{ kind: "EXPIRED"; }' is not assignable to parameter of type 'never'.
```

That turns "forgot to handle a new state" from a production bug into a compile error. `noFallthroughCasesInSwitch` adds a second safety net: a non-empty `case` that forgets to `return`/`break` is an error. Grouping empty cases (`case "CANCELLED": case "REJECTED":`) is still allowed.

> **How to say it in an interview:** "I model order status as a discriminated union so each state only carries valid data — you can't read `avgPrice` on an open order. The `switch` ends with `assertNever`, so adding a new state forces me to handle it everywhere."
